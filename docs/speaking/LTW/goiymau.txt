II. BẢNG TRA CỨU NHANH KIỂU DỮ LIỆU TRONG models.py (CÂU 1 - 3.0 ĐIỂM)
--------------------------------------------------------------------------------
- Chuỗi ký tự (N)          -> models.CharField(max_length=N)
- Văn bản dài / Mô tả      -> models.TextField()
- Số nguyên                -> models.IntegerField()
- Số thập phân / Giá tiền  -> models.DecimalField(max_digits=10, decimal_places=2)
- Ngày tháng               -> models.DateField()
- Ngày và giờ              -> models.DateTimeField()
- Email                    -> models.EmailField()
- Đúng / Sai               -> models.BooleanField(default=False)

* Các thuộc tính đi kèm trong ngoặc:
  + Không trùng lặp        -> unique=True       (VD: models.EmailField(unique=True))
  + Bắt buộc nhập          -> blank=False       (Mặc định của Django là bắt buộc)
  + Không bắt buộc / Trống -> blank=True, null=True
  + Khóa ngoại (1 - N)     -> models.ForeignKey(BangCha, on_delete=models.CASCADE, related_name="...")
  + Nhiều - Nhiều (N - N)  -> models.ManyToManyField(BangKia, related_name="...")
  + N-N có bảng trung gian -> models.ManyToManyField(BangKia, through="BangTrungGian")

* Luôn có hàm __str__ ở cuối mỗi bảng:
    def __str__(self):
        return self.name   # (hoặc self.title, self.full_name tùy tên trường chính)

from django import forms
from django.core.exceptions import ValidationError
from django.utils import timezone
from .models import *

class TenForm(forms.ModelForm):
    # Nếu đề yêu cầu thêm 1 ô tích xác nhận không nằm trong bảng CSDL:
    confirm_information = forms.BooleanField(
        required=True,
        label="Xác nhận thông tin",
        error_messages={"required": "Bạn phải xác nhận thông tin."}
    )

    class Meta:
        model = BangGiaoDich
        fields = ["truong_1", "truong_2", "truong_khoa_ngoai"] # Hoặc "__all__"

    def clean(self):
        cleaned_data = super().clean()
        ten = cleaned_data.get("truong_ten", "")
        email = cleaned_data.get("truong_email", "")
        so_luong = cleaned_data.get("so_luong")
        ngay = cleaned_data.get("truong_ngay")
        cha = cleaned_data.get("truong_khoa_ngoai")

        # MẪU 1: Kiểm tra độ dài chuỗi (VD: ít nhất 4 ký tự)
        if ten and len(ten.strip()) < 4:
            raise ValidationError("Tên phải có ít nhất 4 ký tự.")

        # MẪU 2: Kiểm tra ngày tháng so với ngày hiện tại
        today = timezone.now().date()
        if cha and cha.start_date <= today:
            raise ValidationError("Đã quá hạn hoặc đã khai giảng, không thể đăng ký.")

        # MẪU 3: Kiểm tra số lượng tối đa / tồn kho (max_students / stock)
        if cha and cha.enrollments.count() >= cha.max_students:
            raise ValidationError("Đã đạt số lượng tối đa.")

        # MẪU 4: Kiểm tra số âm hoặc bằng 0
        if so_luong is not None and so_luong <= 0:
            raise ValidationError("Số lượng phải lớn hơn 0.")

        # MẪU 5: Kiểm tra trùng lặp dữ liệu trong cùng 1 khóa/đối tượng (.exists())
        if cha and email:
            if BangGiaoDich.objects.filter(course=cha, student_email__iexact=email).exists():
                raise ValidationError("Dữ liệu này đã tồn tại/đăng ký trùng!")

        return cleaned_data

from django.db import models


class Course(models.Model):
    name = models.CharField(max_length=200, blank=False)
    start_date = models.DateField()
    venue = models.CharField(max_length=150)
    max_students = models.IntegerField()

    def __str__(self):
        return self.name


class Instructor(models.Model):
    full_name = models.CharField(max_length=100)
    email = models.EmailField(unique=True)
    expertise = models.CharField(max_length=150)

    def __str__(self):
        return self.full_name


class Lesson(models.Model):
    title = models.CharField(max_length=200)
    duration = models.IntegerField()
    course = models.ForeignKey(
        Course,
        on_delete=models.CASCADE,
        related_name="lessons"
    )
    instructors = models.ManyToManyField(
        Instructor,
        related_name="lessons"
    )

    def __str__(self):
        return self.title


class Enrollment(models.Model):
    student_name = models.CharField(max_length=100)
    student_email = models.EmailField()
    enrolled_date = models.DateField()
    course = models.ForeignKey(
        Course,
        on_delete=models.CASCADE,
        related_name="enrollments"
    )

    def __str__(self):
        return f"{self.student_name} - {self.course.name}"

from django import forms
from django.core.exceptions import ValidationError
from django.utils import timezone
from .models import Enrollment


class CourseEnrollmentForm(forms.ModelForm):
    confirm_information = forms.BooleanField(
        required=True,
        label="Xác nhận thông tin",
        error_messages={
            "required": "Bạn phải tích chọn xác nhận thông tin trước khi đăng ký."
        }
    )

    class Meta:
        model = Enrollment
        fields = ["student_name", "student_email", "course"]
        labels = {
            "student_name": "Họ tên học viên",
            "student_email": "Email học viên",
            "course": "Khóa đào tạo",
        }

    def clean_student_name(self):
        name = self.cleaned_data.get("student_name", "").strip()
        if len(name) < 4:
            raise ValidationError("Họ tên học viên phải có ít nhất 4 ký tự.")
        return name

    def clean(self):
        cleaned_data = super().clean()
        course = cleaned_data.get("course")
        student_email = cleaned_data.get("student_email")

        if course:
            today = timezone.now().date()

            if course.start_date <= today:
                raise ValidationError("Không thể đăng ký vì khóa đào tạo này đã khai giảng.")

            if course.enrollments.count() >= course.max_students:
                raise ValidationError("Khóa đào tạo này đã đạt số lượng học viên tối đa.")

            if student_email:
                trung_email = Enrollment.objects.filter(
                    course=course,
                    student_email__iexact=student_email
                ).exists()
                if trung_email:
                    raise ValidationError("Email này đã đăng ký khóa đào tạo này rồi.")

        return cleaned_data

from django.shortcuts import render, redirect
from django.contrib import messages
from django.utils import timezone
from .models import Course, Instructor
from .forms import CourseEnrollmentForm


def enroll_view(request):
    if request.method == "POST":
        form = CourseEnrollmentForm(request.POST)
        if form.is_valid():
            enrollment = form.save(commit=False)
            enrollment.enrolled_date = timezone.now().date()
            enrollment.save()

            messages.success(
                request,
                f"Đã đăng ký khóa học thành công cho học viên {enrollment.student_name}!"
            )
            return redirect("enroll")
    else:
        form = CourseEnrollmentForm()

    return render(request, "training/enroll.html", {"form": form})


def course_report(request):
    courses = Course.objects.all().order_by("start_date")

    tu_ngay = request.GET.get("tu_ngay", "").strip()
    den_ngay = request.GET.get("den_ngay", "").strip()
    chuyen_mon = request.GET.get("chuyen_mon", "").strip()

    if tu_ngay:
        courses = courses.filter(start_date__gte=tu_ngay)

    if den_ngay:
        courses = courses.filter(start_date__lte=den_ngay)

    if chuyen_mon:
        courses = courses.filter(
            lessons__instructors__expertise__icontains=chuyen_mon
        ).distinct()

    danh_sach_ket_qua = []
    tong_luot_dang_ky = 0

    for course in courses:
        ds_giang_vien = Instructor.objects.filter(lessons__course=course).distinct()
        so_hoc_vien = course.enrollments.count()
        tong_luot_dang_ky += so_hoc_vien

        danh_sach_ket_qua.append({
            "course": course,
            "instructors": ds_giang_vien,
            "student_count": so_hoc_vien,
        })

    context = {
        "danh_sach_ket_qua": danh_sach_ket_qua,
        "tong_khoa_hoc": len(danh_sach_ket_qua),
        "tong_luot_dang_ky": tong_luot_dang_ky,
        "tu_ngay": tu_ngay,
        "den_ngay": den_ngay,
        "chuyen_mon": chuyen_mon,
    }
    return render(request, "training/report.html", context)
